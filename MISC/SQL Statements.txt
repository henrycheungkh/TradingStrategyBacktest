-- Check for ticker without price recently

SELECT ticker, max(Datetime) AS MAX_Datetime FROM `fdata_price_dayend` GROUP BY ticker HAVING max(Datetime) < '2021-01-15' ORDER BY max(Datetime) asc

-- Check XLON Tickers to Fill HL

SELECT AA.* FROM
(SELECT A.* FROM (SELECT * FROM `fdata_tickers` WHERE Market = 'XLON') A LEFT JOIN (SELECT * FROM `fdata_tickers_altid` WHERE AltID_Type = 'HL') B ON A.`Ticker` = B.`Ticker` WHERE B.`Ticker` IS NULL) AA INNER JOIN (SELECT * FROM `fdata_price_dayend` WHERE Datetime = '2021-03-26') BB ON AA.Ticker = BB.ticker
